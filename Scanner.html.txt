            <div class="bg-white/20 w-16 h-16 rounded-full flex items-center justify-center mx-auto mb-3 backdrop-blur-sm">
                <i class="fas fa-qrcode text-3xl"></i>
            </div>
            <h1 class="text-2xl font-bold tracking-tight">Live Scanner</h1>
            <p class="text-blue-100 text-sm mt-1">Arahkan kamera ke QR Code Siswa</p>
        </div>

        <div class="p-6 flex flex-col items-center bg-slate-50 flex-1">
            <div id="reader" class="w-full rounded-2xl overflow-hidden shadow-inner border-4 border-white bg-black mb-6"></div>

            <div id="resultBox" class="hidden w-full p-5 rounded-2xl text-center shadow-md transition-all mb-4"></div>

            <button onclick="startScanner()" id="btnStart" class="w-full bg-blue-600 text-white font-bold py-4 rounded-xl shadow-lg hover:bg-blue-700 transition-all flex justify-center items-center active:scale-95">
                <i class="fas fa-camera mr-2"></i> Aktifkan Kamera
            </button>
            <button onclick="stopScanner()" id="btnStop" class="hidden w-full bg-red-500 text-white font-bold py-4 rounded-xl shadow-lg hover:bg-red-600 transition-all flex justify-center items-center active:scale-95">
                <i class="fas fa-stop mr-2"></i> Matikan Kamera
            </button>
        </div>
    </div>

    <script>
        // ==========================================
        // https://script.google.com/macros/s/AKfycbxH3NySHx35RvrN4iA8ex9zsug6hmk6O4bdz-eFQjo4HKJTeoQeQdP0CTC57c8ltTah/exec
        // ==========================================
        const APPS_SCRIPT_URL = "MASUKKAN_URL_WEB_APP_DI_SINI";

        let html5QrcodeScanner;
        let isProcessing = false;

        function startScanner() {
            if(APPS_SCRIPT_URL === "MASUKKAN_URL_WEB_APP_DI_SINI") {
                Swal.fire("Peringatan", "Anda belum memasukkan URL Web App Apps Script di dalam kode HTML!", "warning");
                return;
            }

            document.getElementById('btnStart').classList.add('hidden');
            document.getElementById('btnStop').classList.remove('hidden');

            html5QrcodeScanner = new Html5Qrcode("reader");
            
            // Konfigurasi performa kamera (kamera belakang)
            const config = { 
                fps: 10, 
                qrbox: { width: 250, height: 250 },
                aspectRatio: 1.0
            };

            html5QrcodeScanner.start(
                { facingMode: "environment" }, 
                config,
                onScanSuccess,
                onScanFailure
            ).catch(err => {
                Swal.fire("Akses Ditolak", "Gagal mengakses kamera. Pastikan Anda memberikan izin.", "error");
                stopScanner();
            });
        }

        function stopScanner() {
            if (html5QrcodeScanner) {
                html5QrcodeScanner.stop().then(() => {
                    document.getElementById('btnStart').classList.remove('hidden');
                    document.getElementById('btnStop').classList.add('hidden');
                    document.getElementById('resultBox').classList.add('hidden');
                }).catch(err => console.error(err));
            }
        }

        function onScanSuccess(decodedText) {
            if (isProcessing) return;
            isProcessing = true; // Kunci sementara agar tidak double scan

            // Mainkan suara Beep
            new Audio('https://www.soundjay.com/buttons/beep-07a.mp3').play().catch(() => {});

            const resBox = document.getElementById('resultBox');
            resBox.classList.remove('hidden');
            resBox.className = "w-full p-5 rounded-2xl text-center font-bold text-lg bg-blue-50 text-blue-800 border border-blue-200 animate-pulse";
            resBox.innerHTML = `<i class="fas fa-spinner fa-spin mr-2"></i> Memeriksa: ${decodedText}...`;

            // Kirim request ke Google Apps Script (Backend) tanpa blokir CORS
            fetch(APPS_SCRIPT_URL, {
                method: 'POST',
                headers: { 'Content-Type': 'text/plain;charset=utf-8' },
                body: JSON.stringify({ action: 'prosesPresensi', nisn: decodedText })
            })
            .then(response => response.json())
            .then(res => {
                if (res.success) {
                    resBox.className = "w-full p-5 rounded-2xl text-center font-bold text-xl bg-emerald-50 text-emerald-800 border border-emerald-200";
                    resBox.innerHTML = `<i class="fas fa-check-circle text-emerald-600 mr-2"></i> ${res.data.nama} <br><span class="text-sm font-normal text-emerald-700 mt-1 block">Hadir Pukul ${res.time}</span>`;
                } else {
                    if (res.message.toLowerCase().includes('sudah')) {
                        resBox.className = "w-full p-5 rounded-2xl text-center font-bold text-lg bg-amber-50 text-amber-800 border border-amber-200";
                        resBox.innerHTML = `<i class="fas fa-exclamation-triangle text-amber-600 mr-2"></i> ${res.message}`;
                    } else {
                        resBox.className = "w-full p-5 rounded-2xl text-center font-bold text-lg bg-red-50 text-red-800 border border-red-200";
                        resBox.innerHTML = `<i class="fas fa-times-circle text-red-600 mr-2"></i> ${res.message}`;
                    }
                }
                // Kamera otomatis siap scan anak berikutnya setelah 2.5 detik
                setTimeout(() => { isProcessing = false; resBox.classList.add('hidden'); }, 2500);
            })
            .catch(err => {
                resBox.className = "w-full p-5 rounded-2xl text-center font-bold text-lg bg-red-50 text-red-800 border border-red-200";
                resBox.innerHTML = `<i class="fas fa-wifi text-red-600 mr-2"></i> Koneksi Server Gagal!`;
                setTimeout(() => { isProcessing = false; resBox.classList.add('hidden'); }, 3000);
            });
        }

        function onScanFailure(error) { /* Abaikan */ }
    </script>
